b)  slli x5, t0, 2      # x5 = (4 * 4)
    add x5, s0, t1      # x5 = arr + (4 * 4)
    lw x6, 0(x5)        # t3 = arr[4]
    addi x6, x6, 1      # t3 = t3 + 1
    sw x6, 0(x5)        # arr[4] = t3

c)  lw t0, 0(s0)        # t0 = arr[0] = 00000001
    xori t0, t0, 0xFF   # t0 = 00000001 ^ 11111111 = 11111110
    addi t0, t0, 1      # t0 = 11111110 + 00000001 = 11111111
